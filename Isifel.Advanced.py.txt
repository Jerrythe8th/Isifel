def unload_image_model():

    global pipe

    if pipe is None:
        return

    print()
    print("🧹 Freeing Stable Diffusion memory...")

    try:

        if device == "cuda":

            pipe.to("cpu")

    except Exception as e:

        print(
            "⚠️ Could not move Stable Diffusion to CPU:",
            e
        )

    try:

        del pipe

    except:

        pass

    pipe = None

    gc.collect()

    if torch.cuda.is_available():

        try:
            torch.cuda.empty_cache()
            torch.cuda.ipc_collect()
        except:
            pass

    print("✅ Stable Diffusion memory released.")

def create_pattern():

    global current_prompt

    n = point_count.value

    x = np.random.uniform(
        0,
        100,
        n
    )

    y = np.random.uniform(
        0,
        100,
        n
    )

    order = np.random.permutation(n)

    random_words = words.copy()

    random.shuffle(
        random_words
    )

    sentence = (
        " ".join(
            random_words[
                :sentence_length.value
            ]
        ).capitalize()
        + "."
    )

    current_prompt = sentence

    prompt.value = sentence

    return x, y, order

def show_pattern():

    with dots_output:

        clear_output(
            wait=True

        x, y, order = create_pattern()

        plt.figure(
            figsize=(8,8)
        )

        plt.plot(
            x[order],
            y[order],
            "-",
            linewidth=line_width.value
        )

        plt.scatter(
            x,
            y,
            s=point_size.value
        )

        if show_numbers.value:

            for i, (px, py) in enumerate(
                zip(x, y),
                1
            ):

                plt.text(
                    px + .7,
                    py + .7,
                    str(i),
                    fontsize=6
                )

        plt.xlim(0,100)
        plt.ylim(0,100)

        if show_grid.value:

            plt.grid(
                True,
                alpha=.25
            )

        else:

            plt.grid(False)

        plt.gca().set_aspect(
            "equal"
        )

        plt.title(
            f"Random Pattern • {point_count.value} Points"
        )

        plt.show()

def generate_image():

    global current_image
    global pipe

    if pipe is None:

        # If video generation unloaded SD,
        # reload it only when needed.

        set_status(
            "LOADING IMAGE MODEL...",
            ORANGE
        )

        try:

            if device == "cuda":

                pipe = StableDiffusionPipeline.from_pretrained(
                    "runwayml/stable-diffusion-v1-5",
                    torch_dtype=torch.float16,
                    safety_checker=None
                )

                pipe = pipe.to("cuda")

            else:

                pipe = StableDiffusionPipeline.from_pretrained(
                    "runwayml/stable-diffusion-v1-5",
                    torch_dtype=torch.float32,
                    safety_checker=None
                )

                pipe = pipe.to("cpu")

        except Exception as e:

            set_status(
                "IMAGE MODEL FAILED",
                RED
            )

            with image_output:

                clear_output(wait=True)

                print(e)

            return

    text = prompt.value.strip()

    if not text:

        set_status(
            "ENTER A PROMPT FIRST",
            RED
        )

        return

    set_status(
        "GENERATING IMAGE...",
        ORANGE
    )

    with image_output:

        clear_output(
            wait=True
        )

        ai_prompt = (
            text
            + ", "
            + style.value
            + ", highly detailed, cinematic lighting, "
              "professional composition, detailed environment"
        )

        seed = seed_box.value

        if seed < 0:

            used_seed = random.randint(
                0,
                999999999
            )

        else:

            used_seed = seed

        generator = torch.Generator(
            device=device
        ).manual_seed(
            used_seed
        )

        try:

            result = pipe(
                ai_prompt,
                negative_prompt=negative_prompt.value,
                num_inference_steps=steps.value,
                guidance_scale=guidance.value,
                width=resolution.value,
                height=resolution.value,
                generator=generator
            )

            current_image = result.images[0]

            display(
                current_image
            )

            history.append({
                "time":
                    datetime.now().strftime(
                        "%H:%M:%S"
                    ),

                "prompt":
                    text,

                "seed":
                    used_seed,

                "image":

# ============================================================
# ISIFEL RESPONSE
# ============================================================

def generate_isifel_response(message):

    if (
        isifel_model is None
        or isifel_tokenizer is None
    ):

        return (
            "⚠️ Isifel is not loaded.\n\n"
            "Check the model-loading messages above."
        )

    studio_context = current_prompt

    system_message = """
You are Isifel, the built-in AI assistant.

Your name is Isifel.

You are part of a creative application called Isifel.

You can help the user with:

- creative ideas
- stories
- image prompts
- connect-the-dots concepts
- coding
- brainstorming
- explanations
- game development
- Roblox
- Minecraft
- creative projects
- general questions

The application can provide you with a randomly generated
creative concept.

Use that concept when relevant, but do not force it into
every answer.

Be helpful, creative, friendly, and conversational.

Your name is Isifel.

Do not call yourself "AI Creative Studio".
"""

    if studio_context:

        system_message += f"""

Current Isifel generated creative concept:

"{studio_context}"
"""

    messages = [
        {
            "role":
                "system",

            "content":
                system_message
        }
    ]

    for speaker, text in chat_history[-12:]:

        if speaker == "You":

            messages.append({
                "role":
                    "user",

                "content":
                    text
            })

        elif speaker == "Isifel":

            messages.append({
                "role":
                    "assistant",

                "content":
                    text
            })

    try:

        formatted_prompt = (
            isifel_tokenizer.apply_chat_template(
                messages,
                tokenize=False,
                add_generation_prompt=True
            )
        )

        inputs = isifel_tokenizer(
            formatted_prompt,
            return_tensors="pt",
            truncation=True,
            max_length=2048
        )

        model_input_device = (
            isifel_model
            .get_input_embeddings()
            .weight
            .device
        )

        inputs = {
            key:
                value.to(model_input_device)

            for key, value
            in inputs.items()
        }

        with torch.no_grad():

            outputs = isifel_model.generate(
                **inputs,
                max_new_tokens=220,
                temperature=.7,
                top_p=.9,
                do_sample=True,
                repetition_penalty=1.05,
                pad_token_id=(
                    isifel_tokenizer.eos_token_id
                )
            )

        input_length = (
            inputs["input_ids"].shape[1]
        )

        new_tokens = outputs[
            0,
            input_length:
        ]

        response = (
            isifel_tokenizer.decode(
                new_tokens,
                skip_special_tokens=True
            )
            .strip()
        )

        if not response:

            response = (
                "I didn't generate a response. "
                "Try asking again!"
            )

        return response

    except Exception as e:

        return (
            "❌ Isifel generation error:\n\n"
            + str(e)
        )

def create_video(_=None):

    global current_video
    global current_image

    if current_image is None:

        set_status(
            "GENERATE AN IMAGE FIRST",
            RED
        )

        with video_output:

            clear_output(wait=True)

            print(
                "Generate an image in the Create tab first."
            )

        return

    if device != "cuda":

        set_status(
            "CUDA GPU REQUIRED",
            RED
        )

        return

    # --------------------------------------------------------
    # Make sure image is in a supported format
    # --------------------------------------------------------

    base = current_image.convert(
        "RGB"
    )

    # --------------------------------------------------------
    # CogVideoX I2V dimensions.
    #
    # Current Diffusers documentation specifies that I2V
    # dimensions must be divisible by 16.
    #
    # Use a conservative 768 x 480 image.
    # --------------------------------------------------------

    target_width = 768
    target_height = 480

    base = base.resize(
        (
            target_width,
            target_height
        ),
        Image.Resampling.LANCZOS
    )

    # --------------------------------------------------------
    # Load CogVideoX only now.
    # --------------------------------------------------------

    if cogvideo_pipe is None:

        set_status(
            "LOADING COGVIDEOX...",
            ORANGE
        )

        with video_output:

            clear_output(wait=True)

            display(
                HTML(
                    """
                    <h3>🎬 Loading CogVideoX...</h3>

                    <p style="
                        color:#8b95a5;
                    ">
                    The first run can take a while because
                    the model is very large.
                    </p>

                    <p style="
                        color:#60a5fa;
                    ">
                    This is a one-time download for this
                    Colab runtime/cache.
                    </p>
                    """
                )
            )

        if not load_cogvideo():

            return

    # --------------------------------------------------------
    # Generate
    # --------------------------------------------------------

    set_status(
        "GENERATING AI VIDEO...",
        ORANGE
    )

    with video_output:

        clear_output(wait=True)

        display(
            HTML(
                """
                <h3>🎬 Generating Isifel Video...</h3>

                <p style="color:#8b95a5;">
                CogVideoX is rendering the animation.
                This can take several minutes on a
                smaller GPU.
                </p>
                """
            )
        )

    text = prompt.value.strip()

    if not text:

        text = (
            "A cinematic animation based on the image, "
            "smooth natural movement, detailed environment"
        )

    ai_prompt = (
        text
        + ", "
        + style.value
        + ", cinematic motion, "
          "smooth movement, consistent subject, "
          "high detail"
    )

    if video_seed.value < 0:

        used_seed = random.randint(
            0,
            999999999
        )

    else:

        used_seed = video_seed.value

    generator = torch.Generator(
        device="cuda"
    ).manual_seed(
        used_seed
    )

    try:

        # ----------------------------------------------------
        # 49 frames is substantially smaller than the
        # standard 81-frame option and is better suited to
        # limited VRAM.
        # ----------------------------------------------------

        result = cogvideo_pipe(
            prompt=ai_prompt,
            image=base,
            num_videos_per_prompt=1,
            num_inference_steps=video_steps.value,
            num_frames=49,
            guidance_scale=video_guidance.value,
            generator=generator
        )

        frames = result.frames[0]

        filename = (
            "/content/isifel_cogvideo_"
            + datetime.now().strftime(
                "%H%M%S"
            )
            + ".mp4"
        )

        # ----------------------------------------------------
        # Export
        # ----------------------------------------------------

        export_to_video(
            frames,
            filename,
            fps=8
        )

        if not os.path.exists(filename):

            raise RuntimeError(
                "Video file was not created."
            )

        file_size = os.path.getsize(
            filename
        )

        if file_size < 1000:

            raise RuntimeError(
                "Video file is too small."
            )

        current_video = filename

        with video_output:

            clear_output(wait=True)

            display(
                HTML(
                    """
                    <h3>🎬 Isifel Video</h3>
                    """
                )
            )

            display(
                Video(
                    filename,
                    embed=True,
                    width=700
                )
            )

            display(
                HTML(
                    f"""
                    <p style="
                        color:#8b95a5;
                    ">
                    ✅ Video created successfully.
                    </p>

                    <p style="
                        color:#8b95a5;
                    ">
                    Seed: {used_seed}
                    </p>

                    <p style="
                        color:#8b95a5;
                    ">
                    File: {html.escape(filename)}
                    </p>
                    """
                )
            )

        set_status(
            "VIDEO READY",
            GREEN
        )

    except torch.cuda.OutOfMemoryError:

        print(
            "❌ CUDA OUT OF MEMORY"
        )

        print()
        print(
            "CogVideoX still needs more VRAM for this run."
        )

        print(
            "The model was loaded with sequential CPU "
            "offloading, which is the lowest-VRAM "
            "Diffusers mode."
        )

        set_status(
            "VIDEO NEEDS MORE VRAM",
            RED
        )

        try:

            torch.cuda.empty_cache()

        except:

            pass

    except Exception as e:

        set_status(
            "VIDEO ERROR",
            RED
        )

        with video_output:

            clear_output(wait=True)

            print(
                "❌ Video generation error:"
            )

            print()
            print(e)
