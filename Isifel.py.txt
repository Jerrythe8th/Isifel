def create_pattern():

    global current_prompt

    n = point_count.value

    x = np.random.uniform(
        0,
        100,
        n
    )

    y = np.random.uniform(
        0,
        100,
        n
    )

    order = np.random.permutation(n)

    random_words = words.copy()

    random.shuffle(
        random_words
    )

    sentence = (
        " ".join(
            random_words[
                :sentence_length.value
            ]
        ).capitalize()
        + "."
    )

    current_prompt = sentence

    prompt.value = sentence

    return x, y, order

def show_pattern():

    with dots_output:

        clear_output(

        x, y, order = create_pattern()

        plt.figure(
            figsize=(8,8)
        )

        plt.plot(
            x[order],
            y[order],
            "-",
            linewidth=line_width.value
        )

        plt.scatter(
            x,
            y,
            s=point_size.value
        )

        if show_numbers.value:

            for i, (px, py) in enumerate(
                zip(x, y),
                1
            ):

                plt.text(
                    px + .7,
                    py + .7,
                    str(i),
                    fontsize=6
                )

        plt.xlim(
            0,
            100
        )

        plt.ylim(
            0,
            100
        )

        if show_grid.value:

            plt.grid(
                True,
                alpha=.25
            )

        else:

            plt.grid(
                False
            )

        plt.gca().set_aspect(
            "equal"
        )

        plt.title(
            f"Random Pattern • {point_count.value} Points"
        )

        plt.show()

def generate_image():

    global current_image

    if pipe is None:

        set_status(
            "IMAGE MODEL NOT LOADED",
            RED
        )

        return

    text = prompt.value.strip()

    if not text:

        set_status(
            "ENTER A PROMPT FIRST",
            RED
        )

        return

    set_status(
        "GENERATING IMAGE...",
        ORANGE
    )

    with image_output:

        clear_output(
            wait=True
        )

        ai_prompt = (
            text
            + ", "
            + style.value
            + ", highly detailed, cinematic lighting, "
              "professional composition, detailed environment"
        )

        seed = seed_box.value

        if seed < 0:

            used_seed = random.randint(
                0,
                999999999
            )

        else:

            used_seed = seed

        generator = torch.Generator(
            device=device
        ).manual_seed(
            used_seed
        )

        try:

            result = pipe(
                ai_prompt,
                negative_prompt=negative_prompt.value,
                num_inference_steps=steps.value,
                guidance_scale=guidance.value,
                width=resolution.value,
                height=resolution.value,
                generator=generator
            )

            current_image = result.images[0]

            display(
                current_image
            )

            history.append({
                "time": datetime.now().strftime("%H:%M:%S"),
                "prompt": text,
                "seed": used_seed,
                "image": current_image
            })

        except Exception as e:

            print(
                "❌ Image generation error:"
            )

            print(
                str(e)
            )

            set_status(
                "IMAGE ERROR",
                RED
            )

            return

    set_status(
        "IMAGE READY",
        GREEN
    )

# BUILT-IN ISIFEL AI CHAT
# ============================================================

def generate_isifel_response(message):

    if isifel_model is None or isifel_tokenizer is None:

        return (
            "⚠️ Isifel is not loaded.\n\n"
            "Check the model-loading messages above "
            "for the actual reason."
        )

    studio_context = current_prompt

    system_message = """
You are Isifel, the built-in AI assistant.

Your name is Isifel.

You are part of a creative application called Isifel.

You can help the user with:

- creative ideas
- stories
- image prompts
- connect-the-dots concepts
- coding
- brainstorming
- explanations
- game development
- Roblox
- Minecraft
- creative projects
- general questions

The application can provide you with a randomly generated
creative concept.

Use that concept when it is relevant, but do not force it
into every answer.

Be helpful, creative, friendly, and conversational.

Your name is Isifel.

Do not call yourself "AI Creative Studio".
Do not call yourself "AI Creative Studio assistant".
"""

    if studio_context:

        system_message += f"""

Current Isifel generated creative concept:

"{studio_context}"
"""

    messages = [
        {
            "role": "system",
            "content": system_message
        }
    ]

    for speaker, text in chat_history[-12:]:

        if speaker == "You":

            messages.append({
                "role": "user",
                "content": text
            })

        elif speaker == "Isifel":

            messages.append({
                "role": "assistant",
                "content": text
            })

    try:

        formatted_prompt = (
            isifel_tokenizer.apply_chat_template(
                messages,
                tokenize=False,
                add_generation_prompt=True
            )
        )

        inputs = isifel_tokenizer(
            formatted_prompt,
            return_tensors="pt",
            truncation=True,
            max_length=2048
        )

        model_input_device = (
            isifel_model
            .get_input_embeddings()
            .weight
            .device
        )

        inputs = {
            key: value.to(
                model_input_device
            )
            for key, value in inputs.items()
        }

        with torch.no_grad():

            outputs = isifel_model.generate(
                **inputs,
                max_new_tokens=220,
                temperature=0.7,
                top_p=0.9,
                do_sample=True,
                repetition_penalty=1.05,
                pad_token_id=(
                    isifel_tokenizer.eos_token_id
                )
            )

        input_length = (
            inputs["input_ids"].shape[1]
        )

        new_tokens = outputs[
            0,
            input_length:
        ]

        response = (
            isifel_tokenizer.decode(
                new_tokens,
                skip_special_tokens=True
            )
            .strip()
        )

        if not response:

            response = (
                "I didn't generate a response. "
                "Try asking again!"
            )

        return response

    except Exception as e:

        return (
            "❌ Isifel generation error:\n\n"
            + str(e)
        )

# ============================================================
# CHAT FUNCTION
# ============================================================

def chat_send_message(_=None):

    message = chat_input.value.strip()

    if not message:

        return

    chat_history.append(
        (
            "You",
            message
        )
    )

    chat_input.value = ""

    refresh_chat()

    set_status(
        "ISIFEL THINKING...",
        PURPLE
    )

    response = generate_isifel_response(
        message
    )

    chat_history.append(
        (
            "Isifel",
            response
        )
    )

    refresh_chat()

    set_status(
        "ISIFEL READY",
        GREEN
    )

chat_send.on_click(
    chat_send_message
)

# ============================================================
# IMAGE → VIDEO — FIXED
# ============================================================

def create_video(_=None):

    global current_video

    if current_image is None:

        set_status(
            "GENERATE AN IMAGE FIRST",
            RED
        )

        return

    set_status(
        "CREATING VIDEO...",
        ORANGE
    )

    frames = []

    # --------------------------------------------------------
    # Prepare image
    # --------------------------------------------------------

    base = current_image.convert(
        "RGB"
    )

    duration = video_duration.value
    fps = video_fps.value

    total_frames = (
        duration * fps
    )

    width, height = base.size

    # --------------------------------------------------------
    # H.264 / YUV420P prefers even dimensions
    # --------------------------------------------------------

    width = width - (
        width % 2
    )

    height = height - (
        height % 2
    )

    base = base.resize(
        (
            width,
            height
        ),
        Image.Resampling.LANCZOS
    )

    # --------------------------------------------------------
    # Create frames
    # --------------------------------------------------------

    for i in range(total_frames):

        progress = (
            i /
            max(
                total_frames - 1,
                1
            )
        )

        zoom = (
            1 +
            (
                video_motion.value *
                progress
            )
        )

        new_w = int(
            width * zoom
        )

        new_h = int(
            height * zoom
        )

        # Keep resized dimensions even
        new_w = new_w - (
            new_w % 2
        )

        new_h = new_h - (
            new_h % 2
        )

        frame = base.resize(
            (
                new_w,
                new_h
            ),
            Image.Resampling.LANCZOS
        )

        left = (
            new_w - width
        ) // 2

        top = (
            new_h - height
        ) // 2

        frame = frame.crop(
            (
                left,
                top,
                left + width,
                top + height
            )
        )

        frame = frame.convert(
            "RGB"
        )

        frames.append(
            np.asarray(
                frame,
                dtype=np.uint8
            )
        )

    filename = (
        "/content/isifel_video_"
        + datetime.now().strftime(
            "%H%M%S"
        )
        + ".mp4"
    )

    # ========================================================
    # ENCODE MP4
    # ========================================================

    writer = None

    try:

        print(
            "Encoding H.264 video..."
        )

        writer = imageio.get_writer(
            filename,
            format="FFMPEG",
            mode="I",
            fps=fps,
            codec="libx264",
            pixelformat="yuv420p",
            output_params=[
                "-movflags",
                "+faststart",
                "-preset",
                "medium",
                "-crf",
                "23"
            ]
        )

        for frame in frames:

            writer.append_data(
                frame
            )

        writer.close()
        writer = None

        # ----------------------------------------------------
        # Verify file
        # ----------------------------------------------------

        if not os.path.exists(
            filename
        ):

            raise RuntimeError(
                "The MP4 file was not created."
            )

        file_size = os.path.getsize(
            filename
        )

        if file_size < 1000:

            raise RuntimeError(
                "The MP4 file is too small "
                "and may be corrupted."
            )

        current_video = filename

        print(
            "✅ Video encoded successfully!"
        )

        print(
            f"Size: "
            f"{file_size / 1024 / 1024:.2f} MB"
        )

    except Exception as e:

        if writer is not None:

            try:
                writer.close()
            except:
                pass

        set_status(
            "VIDEO ERROR",
            RED
        )

        with video_output:

            clear_output()

            print(
                "❌ Video creation error:"
            )

            print(
                str(e)
            )

        return

    # ========================================================
    # EMBED VIDEO DIRECTLY IN COLAB
    # ========================================================

    with video_output:

        clear_output(
            wait=True
        )

        display(
            HTML(
                """
                <h3 style="
                    margin-bottom:8px;
                ">
                    🎬 Isifel Video
                </h3>
                """
            )
        )

        display(
            Video(
                filename,
                embed=True,
                width=700
            )
        )

        display(
            HTML(
                f"""
                <p style="
                    color:#8b95a5;
                ">
                    ✅ Video created successfully.
                </p>

                <p style="
                    color:#8b95a5;
                ">
                    File: {html.escape(filename)}
                </p>
                """
            )
        )

    set_status(
        "VIDEO READY",
        GREEN
    )

video_button.on_click(

def save_key(_=None):

    name = key_name.value.strip()

    key = api_key.value.strip()

    if not name or not key:

        with keys_output:

            clear_output()

            print(
                "Enter a name and API key."
            )

        return

    saved_keys[name] = {
        "provider": provider.value,
        "key": key
    }

    with keys_output:

        clear_output()

        print(
            f"Saved '{name}' for this runtime."
        )

        print(
            "The key is not written into the Python source."
        )

    api_key.value = ""

def delete_key(_=None):

    name = key_name.value.strip()

    if name in saved_keys:

        del saved_keys[name]

        with keys_output:

            clear_output()

            print(
                f"Deleted '{name}'."
            )

    else:

        with keys_output:

            clear_output()

            print(
                "No saved key with that name."
            )
